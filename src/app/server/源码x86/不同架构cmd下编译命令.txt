#### x86_64架构完整编译命令，cd进入*.go文件所在目录，输入：

set GOPROXY=https://goproxy.cn,direct
go mod init fn-appstores-client
go mod tidy

set CGO_ENABLED=0
set GOOS=linux
set GOARCH=amd64
go build -ldflags="-s -w" -o fn-appstores-client .

dir fn-appstores-client
